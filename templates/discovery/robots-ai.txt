# DEFAULT POSTURE: allow search and retrieval bots, block model-training crawlers.
# This is the recommended starting point — it keeps your docs discoverable and
# citable in AI answers without donating them to training corpora. If your project
# wants training visibility instead, see "TRAINING-VISIBLE ALTERNATIVE" below.
#
# Customize USER-AGENT lines and DISALLOW/ALLOW rules for your project.

# ---------------------------------------------------------------------------
# Content-Signal: state the content-use policy before any per-bot directives.
# Content Signals are a machine-readable statement of intent that sits alongside
# robots.txt rules. Declare them first so crawlers that honour the convention see
# your policy even if they ignore individual user-agent blocks.
#   search    — may index for search results
#   ai-input  — may retrieve at inference time to ground an answer
#   ai-train  — may be used as model training data
# ---------------------------------------------------------------------------
User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=no
Disallow: /admin/
Disallow: /private/
Disallow: /.env*
Disallow: /secrets/
Disallow: /*.json$
Disallow: /*.sql$
Disallow: *.pem
Allow: /llms.txt
Allow: /llms-full.txt
Allow: /robots.txt
Allow: /sitemap.xml
Allow: /AGENTS.md
# Keep this Allow: the Disallow: /*.json$ rule above would otherwise block the
# discovery endpoints (server-card.json, agent-skills index, api-catalog.json).
# Longest-match precedence lets this Allow win for /.well-known/ paths.
Allow: /.well-known/

# Search bots (real-time search indexing) — allowed

Disallow: /private
