## Summary

Detect regressions before production via PR-gated evals. Use layered strategy: Stage 1 (`<1` min, routing tests with mocked tools and code-based graders, Vitest); Stage 2 (1-3 min, real calls, real LLM, baseline comparison vs. main); Stage 3 (optional, 5-10 min nightly, red-team suite). Track statistical significance (gate on pass@k improvements across runs rather than any single run), token budget guardrails (cost per eval case), and baseline regression detection (compare to main-branch results). Fail fast strategy: fail on Stage 1, run Stage 2 only if Stage 1 passes, skip Stage 3 on PRs.

```
Stage 1: Fast routing (`<1` min)
         ↓
Stage 2: Regression suite (1-3 min)
         ↓
Stage 3: Red-team (nightly, optional)
```

---

Agent evals in CI detect regressions before they reach production. The challenge is balancing speed (evals must run in `<5` min) with confidence (evals must catch real regressions).

## Strategy: layered evaluation

Run evals in stages, failing fast:

**Stage 1 (Fast, `<1` min)** - Routing tests with mock tool responses and code-based graders, Vitest harness. Where a live model picks the tool, run each case several times and gate on a pass rate; do not rely on a sampling setting for repeatability, because Claude API version `2026-10-21` removes `temperature`, `top_p` and `top_k`.

**Stage 2 (Medium, 1-3 min)** - Regression suite. Real tool calls, real LLM, against baseline from main branch.

**Stage 3 (Optional, 5-10 min)** - Red-team suite. Runs nightly or on explicit trigger. Too slow for every PR.

```yaml
# .github/workflows/evals.yml
name: Agent Evals

on:
  pull_request:
    branches: [main]
  schedule:
    - cron: "0 3 * * *" # nightly, drives the redteam job below

jobs:
  routing:
    if: github.event_name == 'pull_request'
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - uses: actions/checkout@v7
      - uses: actions/setup-node@v7
        with:
          node-version: 22
      - run: npm ci
      - run: npm run test:routing
        # Vitest harness, mock tools, code-based graders
        # Should complete in `<1` min
      - name: Fail fast on routing errors
        if: failure()
        run: exit 1

  regression:
    if: github.event_name == 'pull_request'
    needs: routing
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v7
      - uses: actions/setup-node@v7
      - run: npm ci

      - name: Download baseline
        run: |
          aws s3 cp s3://evals-baseline/main-latest.json baseline.json
        env:
          AWS_ACCESS_KEY_ID: ${{ secrets.AWS_ACCESS_KEY_ID }}
          AWS_SECRET_ACCESS_KEY: ${{ secrets.AWS_SECRET_ACCESS_KEY }}

      - name: Run regression evals
        run: npx promptfoo eval -c regression-suite.yaml --output current.json
        env:
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
          # promptfoo exits 100 when any test fails. The comparison below is the gate,
          # so let this step finish and keep the output file.
          PROMPTFOO_FAILED_TEST_EXIT_CODE: 0

      - name: Compare to baseline and gate
        run: npm run evals:compare baseline.json current.json
        # Prints pass rate and cost deltas; exits 1 on a regression (script below)

      - name: Comment results on PR
        uses: actions/github-script@v7
        if: always()
        with:
          script: |
            const fs = require('fs');
            const baseline = JSON.parse(fs.readFileSync('baseline.json'));
            const current = JSON.parse(fs.readFileSync('current.json'));

            // Promptfoo's output file keeps counts under results.stats
            const rate = (f) => {
              const s = f.results.stats;
              return s.successes / (s.successes + s.failures + s.errors);
            };
            const baselinePass = (rate(baseline) * 100).toFixed(1);
            const currentPass = (rate(current) * 100).toFixed(1);
            const delta = ((rate(current) - rate(baseline)) * 100).toFixed(1);

            github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: `Eval Results\n\nBaseline: ${baselinePass}%\nCurrent: ${currentPass}%\nDelta: ${delta}%\n\n${delta < -5 ? '⚠️ REGRESSION' : '✅ PASS'}`
            });

  redteam:
    if: github.event_name == 'schedule' || contains(github.event.pull_request.labels.*.name, 'security')
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v7
      - run: npm ci
      # promptfoo exits 100 when any test fails, which fails the job
      - run: npx promptfoo eval -c red-team-suite.yaml --output redteam.json
        env:
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
```

## Pin the dataset version

A baseline-vs-current comparison is only valid if both runs evaluated the same dataset version. If the dataset changed between the baseline run and the PR run, the delta conflates two causes - the code change and the dataset change - and the gate becomes noise. Pin the version per run, record it in the results, and fail fast when a run is unpinned or the two versions disagree. With Promptfoo, put the version under the config's top-level `metadata`; the output file echoes the config back under `config`.

```bash
# Fail the job before interpreting any delta if versions do not match
BASELINE_VERSION=$(jq -r '.config.metadata.datasetVersion // "unpinned"' baseline.json)
CURRENT_VERSION=$(jq -r '.config.metadata.datasetVersion // "unpinned"' current.json)

if [ "$CURRENT_VERSION" = "unpinned" ]; then
  echo "Eval ran against an unpinned dataset. Refusing to gate on it."
  exit 1
fi

if [ "$BASELINE_VERSION" != "$CURRENT_VERSION" ]; then
  echo "Dataset version mismatch: baseline=$BASELINE_VERSION current=$CURRENT_VERSION"
  echo "A dataset change is not a regression. Rebaseline before comparing."
  exit 1
fi
```

Deciding _what_ goes into that versioned dataset - sourcing, labeling, leakage checks, and re-sampling as production drifts - is covered in [Datasets and experiments](/docs/testing/datasets-and-experiments).

## Statistical significance

A single regression is noise. Use statistical tests to distinguish signal from noise.

### Bootstrap confidence intervals

Resample the cases with replacement, and keep the pairing when both systems ran the same cases. A shuffle of the original sample is not a resample: every shuffle has the same pass count, so the interval collapses to the observed difference.

```typescript
// baseline[i] and current[i] are the pass/fail results for the same case i
function pairedBootstrap(baseline: boolean[], current: boolean[], samples = 2000) {
  const n = baseline.length;
  if (current.length !== n) throw new Error("Paired comparison needs the same cases in both runs");

  const diffs: number[] = [];
  for (let s = 0; s < samples; s++) {
    let delta = 0;
    for (let j = 0; j < n; j++) {
      const i = Math.floor(Math.random() * n); // draw a case index with replacement
      delta += Number(current[i]) - Number(baseline[i]);
    }
    diffs.push(delta / n);
  }

  diffs.sort((a, b) => a - b);
  const lower = diffs[Math.floor(samples * 0.025)];
  const upper = diffs[Math.floor(samples * 0.975)];
  console.log(`95% CI on pass-rate difference: [${lower.toFixed(3)}, ${upper.toFixed(3)}]`);

  // If the interval includes 0, there is no significant difference
  if (lower <= 0 && upper >= 0) return "No significant difference";
  return upper < 0 ? "Regression (significant)" : "Improvement";
}
```

### Paired t-test

For continuous scores on the same cases (a judge score, a latency), a paired t-test compares the two runs. For pass/fail outcomes on the same cases, the paired bootstrap above or McNemar's test is the better fit.

```typescript
function pairedT(baseline: number[], current: number[]): { t: number; df: number } {
  // t = mean(differences) / (stdev / sqrt(n)), with n - 1 degrees of freedom
  const diffs = baseline.map((b, i) => current[i] - b);
  const n = diffs.length;
  const mean = diffs.reduce((a, b) => a + b) / n;
  const variance = diffs.reduce((sum, d) => sum + Math.pow(d - mean, 2), 0) / (n - 1);
  const stdev = Math.sqrt(variance);

  return { t: mean / (stdev / Math.sqrt(n)), df: n - 1 };
}

// Turn t and df into a p-value with a stats library: scipy.stats.ttest_rel returns both
// for a paired test. A fixed cutoff is wrong because the critical value depends on n, and
// a regression is a large negative t, so test the direction you care about.
```

## Shadow evaluation

Deploy code to shadow traffic without gating on evals. Collect eval metrics on real production tasks, then gate on those metrics.

```yaml
# Deploy agent v2 to shadow traffic (5% of requests)
# Both v1 and v2 run; only v1's response is used
# v2 runs against mocked or read-only tools; its evals are stored

job_params:
  shadow_model_weight: 0.05

# Collect evals after 24h
aws s3 cp s3://eval-results/shadow-2025-04-17.json .

# Query: pass@1=0.88, tool_f1=0.94
# If both above thresholds, promote v2 to 100%
```

Benefits:

- Users still get v1's response - but suppressing v2's output does not suppress its side effects, so run v2 against isolated, mocked or read-only tools (a shadow agent that can send email or write records still does)
- Real production data (not synthetic test cases)
- Unbiased eval (evaluator doesn't know if it's v1 or v2)

## Cost and latency budgets

Cap eval costs and latencies to prevent runaway expenses:

```typescript
interface EvalBudget {
  maxCostPerEval: number; // $ per eval run
  maxLatencyP99: number; // milliseconds
  maxTokensPerTask: number;
}

async function runEvalWithBudget(budget: EvalBudget) {
  let totalCost = 0;
  let totalTokens = 0;
  const latencies: number[] = [];

  for (const task of testCases) {
    const start = Date.now();

    const result = await agent.run(task);

    const latency = Date.now() - start;
    latencies.push(latency);

    const cost = estimateTokenCost(result.tokensUsed);
    totalCost += cost;
    totalTokens += result.tokensUsed;

    // Check budgets in real time
    if (totalCost > budget.maxCostPerEval) {
      throw new Error(`Cost budget exceeded: $${totalCost}`);
    }

    if (totalTokens > budget.maxTokensPerTask * testCases.length) {
      throw new Error(`Token budget exceeded`);
    }

    if (latency > budget.maxLatencyP99) {
      console.warn(`Single task latency ${latency}ms exceeds budget ${budget.maxLatencyP99}ms`);
    }
  }

  const latencies_sorted = latencies.sort((a, b) => a - b);
  const p99 = latencies_sorted[Math.floor(latencies_sorted.length * 0.99)];

  console.log(`Cost: $${totalCost.toFixed(2)}, P99 latency: ${p99}ms`);
}
```

Set budgets based on previous runs:

```yaml
# If last eval cost $50 and took 5 min, allow ±20% slack
cost_budget: $60
latency_budget: 6min
token_budget: 150k
```

## Slicing by scenario

Separate evals for different scenarios to catch category-specific regressions:

```yaml
tests:
  - category: "happy_path"
    description: "Well-formed intent, standard tools"
    skip_in_ci: false

  - category: "edge_case"
    description: "Malformed input, missing context"
    skip_in_ci: false

  - category: "adversarial"
    description: "Prompt injection, jailbreak attempts"
    skip_in_ci: true # Too slow, run nightly instead

# Report separately
evals_happy_path: 0.95
evals_edge_case: 0.82
evals_adversarial: 0.98
```

## Example workflow

```typescript
// scripts/eval-compare.ts
import { readFileSync } from "fs";

// Subset of Promptfoo's --output file: counts under results.stats, per-case rows under results.results
interface PromptfooOutput {
  results: {
    stats: { successes: number; failures: number; errors: number };
    results: { success: boolean; cost?: number }[];
  };
}

function summarize(file: PromptfooOutput) {
  const { successes, failures, errors } = file.results.stats;
  return {
    pass_rate: successes / (successes + failures + errors),
    cost: file.results.results.reduce((sum, r) => sum + (r.cost ?? 0), 0),
  };
}

const baseline = summarize(JSON.parse(readFileSync("baseline.json", "utf8")));
const current = summarize(JSON.parse(readFileSync("current.json", "utf8")));

const passRateDelta = (current.pass_rate - baseline.pass_rate) * 100;
const costDelta = ((current.cost - baseline.cost) / baseline.cost) * 100;

console.log(
  `Pass rate: ${baseline.pass_rate.toFixed(2)} → ${current.pass_rate.toFixed(2)} (${passRateDelta > 0 ? "+" : ""}${passRateDelta.toFixed(1)}%)`,
);
console.log(
  `Cost: $${baseline.cost.toFixed(2)} → $${current.cost.toFixed(2)} (${costDelta > 0 ? "+" : ""}${costDelta.toFixed(1)}%)`,
);

// Gate on regression
if (passRateDelta < -5) {
  console.error("Regression detected. Please fix before merging.");
  process.exit(1);
}

if (costDelta > 20) {
  console.warn("Cost increased by >20%. Review before merging.");
}
```

Other metrics, such as tool F1, come from named metrics on your assertions (`namedScores` on each result) rather than from a built-in summary field.
